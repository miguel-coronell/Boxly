rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function estaLogueado() {
      return request.auth != null;
    }
    function negocioIdDelUsuario() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data.negocioId;
    }
    function perteneceAlNegocio(negocioId) {
      return estaLogueado() && negocioIdDelUsuario() == negocioId;
    }
    function isCreator() {
      return estaLogueado()
        && request.auth.token.email == "miguelcoronell94@gmail.com"
        && request.auth.token.email_verified == true;
    }
    function esEncargado() {
      return estaLogueado() && request.auth.token.rol == "encargado";
    }
    function miSucursalClaim() {
      return request.auth.token.sucursalId;
    }

    match /{document=**} {
      allow read, write: if isCreator();
    }

    match /users/{uid} {
      allow read: if estaLogueado() && (request.auth.uid == uid || negocioIdDelUsuario() == resource.data.negocioId);
      allow create: if estaLogueado()
        && request.auth.uid == uid
        && request.resource.data.negocioId == uid
        && request.resource.data.rol == "Administrador";
      allow update: if estaLogueado()
        && request.auth.uid == uid
        && request.resource.data.diff(resource.data).affectedKeys()
             .hasOnly(['nombre', 'telefono', 'fotoURL']);
      allow delete: if false;
    }

    match /negocios/{negocioId} {
      allow read, update: if perteneceAlNegocio(negocioId);
      allow create: if estaLogueado() && negocioId == request.auth.uid;
      allow delete: if false;

      match /productos/{productId} {
        allow read, write: if perteneceAlNegocio(negocioId);
      }

      match /movimientos/{movId} {
        allow read: if perteneceAlNegocio(negocioId) &&
          (!esEncargado() || miSucursalClaim() == resource.data.sucursalId);
        allow create: if perteneceAlNegocio(negocioId) &&
          (!esEncargado() || miSucursalClaim() == request.resource.data.sucursalId);
        allow update, delete: if perteneceAlNegocio(negocioId) &&
          (!esEncargado() || miSucursalClaim() == resource.data.sucursalId);
      }

      match /sucursales/{sucursalId} {
        allow read: if perteneceAlNegocio(negocioId);
        allow write: if perteneceAlNegocio(negocioId) && !esEncargado();
      }

      match /usuarios/{usuarioId} {
        allow read, write: if perteneceAlNegocio(negocioId);
      }
    }
  }
}
